# ─────────────────────────────────────────────────────────────────────────────
# CONFIGURATION
# ─────────────────────────────────────────────────────────────────────────────
DATA_DIR    = r"C:\Path\to\dir"
SARC_FILE   = os.path.join(DATA_DIR, "Sarcoma.csv")
NORMAL_FILE = os.path.join(DATA_DIR, "Normal.csv")   # <-- Copy
OUT_DIR     = os.path.join(DATA_DIR, "sarcoma_VS_Normal_2")

MIRNA_COLS = [
    "hsa-mir-196a", "hsa-mir-214", "hsa-mir-767",
    "hsa-mir-1245a", "hsa-mir-1305", "hsa-mir-337", "hsa-mir-493"
]

FDR_THRESHOLD  = 0.05
FC_THRESHOLD   = 0.5      # z-score units (difference of medians)

TOP_N_VIOLIN  = 7
TOP_N_HEATMAP = 7
HEATMAP_ZSCORE = False    # data already z-scored; set True for extra row re-centering

# ─────────────────────────────────────────────────────────────────────────────
# HELPERS
# ─────────────────────────────────────────────────────────────────────────────

def _bh_correction(p_values: np.ndarray) -> tuple:
    """Benjamini-Hochberg FDR correction (numpy-only)."""
    n = len(p_values)
    order = np.argsort(p_values)
    ranks = np.empty(n, dtype=int)
    ranks[order] = np.arange(1, n + 1)
    p_adj = np.minimum(1.0, p_values * n / ranks)
    p_adj_ordered = p_adj[order]
    for i in range(n - 2, -1, -1):
        p_adj_ordered[i] = min(p_adj_ordered[i], p_adj_ordered[i + 1])
    p_adj[order] = p_adj_ordered
    return p_adj < FDR_THRESHOLD, p_adj


def rank_biserial_correlation(u_stat, n1, n2):
    return (2 * u_stat) / (n1 * n2) - 1


def run_mannwhitney(sarc_vals, norm_vals):
    try:
        u_stat, p_val = stats.mannwhitneyu(sarc_vals, norm_vals, alternative="two-sided")
        r = rank_biserial_correlation(u_stat, len(sarc_vals), len(norm_vals))
        return u_stat, p_val, r
    except ValueError:
        return np.nan, 1.0, 0.0


def load_wide_csv(filepath, label):
    """
    Wide-format: rows = samples, first col = Patient_ID, remaining cols = miRNAs.
    Returns DataFrame indexed by Patient_ID, columns = miRNA names only.
    """
    df = pd.read_csv(filepath, index_col=0)
    keep = [c for c in df.columns if c in MIRNA_COLS]
    df   = df[keep].apply(pd.to_numeric, errors="coerce")
    print(f"[{label}] Loaded: {df.shape[0]} samples x {df.shape[1]} miRNAs")
    return df

# ─────────────────────────────────────────────────────────────────────────────
# MAIN
# ─────────────────────────────────────────────────────────────────────────────

def main():
    os.makedirs(OUT_DIR, exist_ok=True)

    print("=" * 65)
    print("SARCOMA (Training/TCGA) vs NORMAL  |  miRNA Differential Expression")
    print("Normal file : Normal_combined_normalized_Copy.csv")
    print("Data type   : pre-normalized z-scores")
    print("=" * 65)

    sarc   = load_wide_csv(SARC_FILE,   "SARCOMA")
    normal = load_wide_csv(NORMAL_FILE, "NORMAL (Copy)")

    # Align to shared miRNAs
    shared = sorted(set(sarc.columns) & set(normal.columns))
    if not shared:
        raise ValueError("No shared miRNA columns found. Check column names.")
    sarc   = sarc[shared]
    normal = normal[shared]

    # NOTE: In the Copy file, hsa-mir-1245a and hsa-mir-1305 are already 0
    # (not NaN), so no imputation is needed. Still handle any residual NaN.
    for col in shared:
        sarc[col].fillna(sarc[col].mean(), inplace=True)
        normal[col].fillna(normal[col].mean(), inplace=True)

    sarc.dropna(axis=1, how="all", inplace=True)
    normal.dropna(axis=1, how="all", inplace=True)
    shared = sorted(set(sarc.columns) & set(normal.columns))
    sarc   = sarc[shared]
    normal = normal[shared]

    n_sarc, n_normal = len(sarc), len(normal)
    print(f"\nShared miRNAs   : {len(shared)}")
    print(f"Sarcoma samples : {n_sarc}")
    print(f"Normal samples  : {n_normal}")

    # Transpose: rows = miRNAs, cols = samples
    sarc_T   = sarc.T
    normal_T = normal.T

    # ── Per-miRNA stats ────────────────────────────────────────────────────────
    print("\n-- Computing per-miRNA statistics...")
    mean_s  = sarc_T.mean(axis=1)
    mean_n  = normal_T.mean(axis=1)
    std_s   = sarc_T.std(axis=1)
    std_n   = normal_T.std(axis=1)
    med_s   = sarc_T.median(axis=1)
    med_n   = normal_T.median(axis=1)
    fc_med  = med_s - med_n
    fc_mean = mean_s - mean_n

    # ── Mann-Whitney + BH FDR ─────────────────────────────────────────────────
    u_list, p_list, r_list = [], [], []
    for mirna in shared:
        u, p, r = run_mannwhitney(
            sarc_T.loc[mirna].values.astype(float),
            normal_T.loc[mirna].values.astype(float)
        )
        u_list.append(u); p_list.append(p); r_list.append(r)

    p_arr = np.array(p_list)
    _, p_adj = _bh_correction(p_arr)

    # ── Results table ─────────────────────────────────────────────────────────
    results = pd.DataFrame({
        "miRNA_ID"        : shared,
        "mean_sarcoma"    : mean_s.values,
        "mean_normal"     : mean_n.values,
        "median_sarcoma"  : med_s.values,
        "median_normal"   : med_n.values,
        "std_sarcoma"     : std_s.values,
        "std_normal"      : std_n.values,
        "FC_median"       : fc_med.values,
        "FC_mean"         : fc_mean.values,
        "U_statistic"     : np.array(u_list),
        "p_value"         : p_arr,
        "p_adj_BH"        : p_adj,
        "rank_biserial_r" : np.array(r_list),
        "significant"     : p_adj < FDR_THRESHOLD,
    }).set_index("miRNA_ID")

    results["direction"] = np.where(
        results["FC_median"] > 0, "Up in Sarcoma",
        np.where(results["FC_median"] < 0, "Down in Sarcoma", "No change")
    )
    results.sort_values("p_adj_BH", inplace=True)

    results.to_csv(os.path.join(OUT_DIR, "DE_full_results.csv"))
    print(f"  Full results saved -> {os.path.join(OUT_DIR, 'DE_full_results.csv')}")

    sig = results[
        (results["p_adj_BH"] < FDR_THRESHOLD) &
        (results["FC_median"].abs() >= FC_THRESHOLD)
    ]
    sig.to_csv(os.path.join(OUT_DIR, "DE_significant_hits.csv"))

    n_up   = (sig["direction"] == "Up in Sarcoma").sum()
    n_down = (sig["direction"] == "Down in Sarcoma").sum()
    print(f"\n-- Significant miRNAs (FDR < {FDR_THRESHOLD}, |FC| >= {FC_THRESHOLD}):")
    print(f"   Total            : {len(sig)}")
    print(f"   Up in sarcoma    : {n_up}")
    print(f"   Down in sarcoma  : {n_down}")
    print(f"  Significant hits saved -> {os.path.join(OUT_DIR, 'DE_significant_hits.csv')}")

    print("\n  All miRNA results:")
    print(
        results[["FC_median", "FC_mean", "p_value", "p_adj_BH", "rank_biserial_r"]]
        .to_string(float_format="{:.4f}".format)
    )

    # ── Plots ─────────────────────────────────────────────────────────────────
    print("\n-- Generating plots...")
    top_n_v = min(TOP_N_VIOLIN,  len(shared))
    top_n_h = min(TOP_N_HEATMAP, len(shared))

    _plot_summary_stats(results, OUT_DIR, FDR_THRESHOLD)
    _plot_volcano(results, OUT_DIR, FDR_THRESHOLD, FC_THRESHOLD)
    _plot_heatmap(sarc_T, normal_T, results, OUT_DIR, top_n_h, HEATMAP_ZSCORE)
    _plot_violins(sarc_T, normal_T, results, OUT_DIR, top_n_v)
    _plot_ma(results, OUT_DIR, FDR_THRESHOLD, FC_THRESHOLD)

    print(f"\nAnalysis complete. All outputs saved to:\n  {OUT_DIR}")
    print("=" * 65)

def _plot_summary_stats(results, out_dir, fdr_thresh):
    fig, axes = plt.subplots(1, 3, figsize=(16, 5))
    fig.suptitle("Summary Statistics -- Sarcoma (Training) vs Normal [Copy]",
                 fontsize=13, y=1.02)

    # Ranked p-values bar chart
    ax = axes[0]
    ranked = results["p_value"].sort_values()
    ax.bar(range(len(ranked)), ranked.values,
           color="#4C72B0", edgecolor="white", linewidth=0.4)
    for i, (mirna, pv) in enumerate(ranked.items()):
        ax.text(i, pv + 0.005, mirna.replace("hsa-mir-", "mir-"),
                fontsize=7, ha="center", va="bottom", rotation=45)
    ax.axhline(fdr_thresh, color="red", linestyle="--", linewidth=1.2,
               label=f"p={fdr_thresh}")
    ax.set_xlabel("Ranked miRNAs")
    ax.set_ylabel("Raw p-value (Mann-Whitney)")
    ax.set_title("P-values (ranked)")
    ax.legend(fontsize=9)

    # FC bar chart
    ax = axes[1]
    colors = ["#D62728" if v > 0 else "#1F77B4" for v in results["FC_median"].values]
    ax.bar(range(len(results)), results["FC_median"].values,
           color=colors, edgecolor="white", linewidth=0.5)
    ax.set_xticks(range(len(results)))
    ax.set_xticklabels(
        [m.replace("hsa-mir-", "mir-") for m in results.index],
        rotation=45, ha="right", fontsize=8
    )
    ax.set_ylabel("Median FC (z-score, Sarcoma - Normal)")
    ax.set_title("Fold Change per miRNA (median)")
    ax.axhline(0, color="black", linewidth=0.8)
    ax.axhline( FC_THRESHOLD, color="orange", linestyle="--", linewidth=1, alpha=0.8)
    ax.axhline(-FC_THRESHOLD, color="orange", linestyle="--", linewidth=1,
               alpha=0.8, label=f"|FC|={FC_THRESHOLD}")
    ax.legend(fontsize=9)

    # Effect size bar chart
    ax = axes[2]
    colors2 = ["#D62728" if v > 0 else "#1F77B4" for v in results["rank_biserial_r"].values]
    ax.bar(range(len(results)), results["rank_biserial_r"].values,
           color=colors2, edgecolor="white", linewidth=0.5)
    ax.set_xticks(range(len(results)))
    ax.set_xticklabels(
        [m.replace("hsa-mir-", "mir-") for m in results.index],
        rotation=45, ha="right", fontsize=8
    )
    ax.set_ylabel("Rank-biserial correlation (r)")
    ax.set_title("Effect Size per miRNA")
    ax.axhline(0, color="black", linewidth=0.8)
    ax.axhline( 0.3, color="grey", linestyle="--", linewidth=1, alpha=0.7, label="r=+/-0.3")
    ax.axhline(-0.3, color="grey", linestyle="--", linewidth=1, alpha=0.7)
    ax.legend(fontsize=9)

    plt.tight_layout()
    path = os.path.join(out_dir, "01_summary_stats.png")
    fig.savefig(path, dpi=180, bbox_inches="tight")
    plt.close(fig)
    print(f"  Plot saved -> {path}")


def _plot_volcano(results, out_dir, fdr_thresh, fc_thresh):
    df = results.copy()
    df["-log10_padj"] = -np.log10(df["p_adj_BH"].clip(lower=1e-300))

    def _cat(row):
        sig = (row["p_adj_BH"] < fdr_thresh) and (abs(row["FC_median"]) >= fc_thresh)
        if not sig: return "NS"
        return "Up" if row["FC_median"] > 0 else "Down"

    df["cat"] = df.apply(_cat, axis=1)
    palette = {"NS": "#BBBBBB", "Up": "#D62728", "Down": "#1F77B4"}
    counts  = df["cat"].value_counts()

    fig, ax = plt.subplots(figsize=(9, 7))
    for cat, colour in palette.items():
        sub = df[df["cat"] == cat]
        ax.scatter(sub["FC_median"], sub["-log10_padj"],
                   c=colour, s=80, alpha=0.8, linewidths=0.5, edgecolors="black",
                   label=f"{cat} (n={counts.get(cat, 0)})")

    ax.axhline(-np.log10(fdr_thresh), color="black", linestyle="--", linewidth=0.9, alpha=0.7)
    ax.axvline( fc_thresh, color="black", linestyle="--", linewidth=0.9, alpha=0.7)
    ax.axvline(-fc_thresh, color="black", linestyle="--", linewidth=0.9, alpha=0.7)

    for mirna, row in df.iterrows():
        ax.text(row["FC_median"], row["-log10_padj"] + 0.03,
                mirna, fontsize=8, ha="center", va="bottom",
                color="black", fontweight="bold")

    ax.set_xlabel("Median FC (z-score units: Sarcoma - Normal)", fontsize=11)
    ax.set_ylabel("-log10 (adjusted p-value)", fontsize=11)
    ax.set_title("Volcano Plot: Sarcoma (Training) vs Normal [Copy]\nmiRNA Expression",
                 fontsize=12)
    ax.legend(title="Category", framealpha=0.8, fontsize=9)
    plt.tight_layout()
    path = os.path.join(out_dir, "02_volcano_plot.png")
    fig.savefig(path, dpi=180, bbox_inches="tight")
    plt.close(fig)
    print(f"  Plot saved -> {path}")


def _plot_heatmap(sarc_T, normal_T, results, out_dir, top_n, z_score):
    top_mirnas = results.head(top_n).index.tolist()
    sarc_sub   = sarc_T.loc[top_mirnas]
    normal_sub = normal_T.loc[top_mirnas]
    combined   = pd.concat([sarc_sub, normal_sub], axis=1)

    n_s, n_n   = sarc_sub.shape[1], normal_sub.shape[1]
    col_colors = ["#D62728"] * n_s + ["#1F77B4"] * n_n

    if z_score:
        combined = combined.apply(lambda row: (row - row.mean()) / (row.std() + 1e-9), axis=1)
        cbar_label, cmap, center = "Row z-score (re-centred)", "RdBu_r", 0
    else:
        cbar_label, cmap, center = "Normalized expression (z-score)", "RdBu_r", 0

    height = max(5, top_n * 0.6)
    g = sns.clustermap(
        combined, col_cluster=False, row_cluster=True,
        col_colors=col_colors, cmap=cmap, center=center,
        figsize=(14, height), yticklabels=True, xticklabels=False,
        linewidths=0, cbar_kws={"label": cbar_label, "shrink": 0.5}
    )
    legend_elements = [
        Patch(facecolor="#D62728", label=f"Sarcoma Training (n={n_s})"),
        Patch(facecolor="#1F77B4", label=f"Normal [Copy] (n={n_n})"),
    ]
    g.ax_heatmap.legend(handles=legend_elements, loc="upper right",
                        bbox_to_anchor=(1.25, 1.05), framealpha=0.8, fontsize=9)
    g.fig.suptitle("miRNA Heatmap  |  Sarcoma (Training) vs Normal [Copy]",
                   y=1.02, fontsize=12)
    path = os.path.join(out_dir, "03_heatmap_mirnas.png")
    g.fig.savefig(path, dpi=150, bbox_inches="tight")
    plt.close(g.fig)
    print(f"  Plot saved -> {path}")


def _plot_violins(sarc_T, normal_T, results, out_dir, top_n):
    top_mirnas = results.head(top_n).index.tolist()
    ncols = min(4, top_n)
    nrows = int(np.ceil(top_n / ncols))
    fig, axes = plt.subplots(nrows, ncols, figsize=(ncols * 4, nrows * 4))
    axes = np.array(axes).flatten()

    for idx, mirna in enumerate(top_mirnas):
        ax    = axes[idx]
        s_vals = sarc_T.loc[mirna].values.astype(float)
        n_vals = normal_T.loc[mirna].values.astype(float)

        plot_df = pd.DataFrame({
            "expression": np.concatenate([s_vals, n_vals]),
            "group": ["Sarcoma"] * len(s_vals) + ["Normal"] * len(n_vals)
        })
        sns.violinplot(data=plot_df, x="group", y="expression",
                       palette={"Sarcoma": "#D62728", "Normal": "#1F77B4"},
                       inner=None, cut=0, linewidth=0.8, ax=ax)
        sns.stripplot(data=plot_df, x="group", y="expression",
                      palette={"Sarcoma": "#8B0000", "Normal": "#00008B"},
                      size=3.5, jitter=True, alpha=0.6, ax=ax)

        row = results.loc[mirna]
        padj_str = f"{row['p_adj_BH']:.2e}" if row['p_adj_BH'] < 0.001 else f"{row['p_adj_BH']:.3f}"
        sig_mk   = "*" if row["p_adj_BH"] < FDR_THRESHOLD else "ns"
        ax.set_title(
            f"{mirna}\nFC={row['FC_median']:+.2f}  FDR={padj_str}  r={row['rank_biserial_r']:.2f}  {sig_mk}",
            fontsize=8.5
        )
        ax.set_xlabel("")
        ax.set_ylabel("Normalized expression (z-score)" if idx % ncols == 0 else "")
        ax.tick_params(labelsize=8)

    for j in range(len(top_mirnas), len(axes)):
        axes[j].set_visible(False)

    fig.suptitle("miRNA Expression Distributions  |  Sarcoma (Training) vs Normal [Copy]",
                 fontsize=12, y=1.01)
    plt.tight_layout()
    path = os.path.join(out_dir, "04_violin_plots.png")
    fig.savefig(path, dpi=150, bbox_inches="tight")
    plt.close(fig)
    print(f"  Plot saved -> {path}")


def _plot_ma(results, out_dir, fdr_thresh, fc_thresh):
    df = results.copy()
    df["A"] = (df["median_sarcoma"] + df["median_normal"]) / 2

    def _cat(row):
        sig = (row["p_adj_BH"] < fdr_thresh) and (abs(row["FC_median"]) >= fc_thresh)
        if not sig: return "NS"
        return "Up" if row["FC_median"] > 0 else "Down"

    df["cat"] = df.apply(_cat, axis=1)
    palette = {"NS": "#BBBBBB", "Up": "#D62728", "Down": "#1F77B4"}
    counts  = df["cat"].value_counts()

    fig, ax = plt.subplots(figsize=(9, 6))
    for cat, colour in palette.items():
        sub = df[df["cat"] == cat]
        ax.scatter(sub["A"], sub["FC_median"],
                   c=colour, s=80, alpha=0.8, linewidths=0.5, edgecolors="black",
                   label=f"{cat} (n={counts.get(cat, 0)})")

    ax.axhline(0,          color="black", linewidth=0.9)
    ax.axhline( fc_thresh, color="black", linestyle="--", linewidth=0.8, alpha=0.6)
    ax.axhline(-fc_thresh, color="black", linestyle="--", linewidth=0.8, alpha=0.6)

    for mirna, row in df.iterrows():
        ax.text(row["A"] + 0.05, row["FC_median"] + 0.04,
                mirna, fontsize=8, color="black")

    ax.set_xlabel("A  (mean normalized z-score, both groups)", fontsize=11)
    ax.set_ylabel("M  (median FC, Sarcoma - Normal, z-score units)", fontsize=11)
    ax.set_title("MA Plot: Sarcoma (Training) vs Normal [Copy]\nmiRNA Expression",
                 fontsize=12)
    ax.legend(title="Category", framealpha=0.8, fontsize=9)
    plt.tight_layout()
    path = os.path.join(out_dir, "05_MA_plot.png")
    fig.savefig(path, dpi=180, bbox_inches="tight")
    plt.close(fig)
    print(f"  Plot saved -> {path}")


# ─────────────────────────────────────────────────────────────────────────────
if __name__ == "__main__":
    main()
